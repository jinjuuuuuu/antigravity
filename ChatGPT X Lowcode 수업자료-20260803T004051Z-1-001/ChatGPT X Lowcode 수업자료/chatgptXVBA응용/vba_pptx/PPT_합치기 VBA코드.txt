Sub MergePPTs()
    Dim MyPresentation As Presentation
    Dim SourcePresentation As Presentation
    Dim SourceSlide As Slide
    Dim i As Integer
    Dim strPath As String
    Dim strFilePattern As String
    Dim strFileName As String

    strPath = "PPT가 모여있는 경로"
    strFilePattern = strPath & "*.pptx"
    strFileName = Dir(strFilePattern)

    '새로운 프레젠테이션 생성
    Set MyPresentation = Presentations.Add

    'new_ppt 폴더에 있는 모든 PPT 파일 병합
    Do While strFileName <> ""
        '각 PPT를 열고 슬라이드를 새 프레젠테이션에 추가
        Set SourcePresentation = Presentations.Open(strPath & strFileName)

        For i = 1 To SourcePresentation.Slides.Count
            Set SourceSlide = SourcePresentation.Slides(i)
            SourceSlide.Copy
            MyPresentation.Slides.Paste
        Next i

        'SourcePresentation을 닫고 다음 파일을 처리
        SourcePresentation.Close
        strFileName = Dir
    Loop

    '병합된 파일을 result.pptx로 저장
    MyPresentation.SaveAs "PPT를 저장하고싶은경로"
    MyPresentation.Close
End Sub
