Sub AdjustSpacingAllShapes()

Dim sld As Slide
Dim shp As Shape

Cycle through all slides
For Each sld In ActivePresentation.Slides
    'Iterate over all shapes on each slide
    For Each shp In sld.Shapes
        'If the shape has a text frame
        If shp.HasTextFrame Then
            'Error handling
            On Error Resume Next
            'Change the font name and character spacing of the text frame
            With shp
                .TextFrame2.TextRange.Font.Spacing = 20
            End With
            'Reset error state
            On Error GoTo 0
        End If
    Next shp
Next sld

End Sub
